            // Get responsive scroll offset based on screen size
            function getScrollOffset() {
                var w = window.innerWidth;
                if (w < 768) return 130;      // Mobile: header ~95px + padding
                if (w < 1024) return 195;     // Tablet: header ~167px + padding
                return baseScrollOffset;       // Desktop: use configured value
            }

            // Position FAB at the right edge of content container
            function positionFab() {
                var content = document.querySelector(contentSelector);
                var fab = document.getElementById('tocFabWrapper_' + id);
                if (!content || !fab) return;
                var rect = content.getBoundingClientRect();
                fab.style.left = (rect.right - fabWidth) + 'px';
            }

            // Toggle dropdown
            window['tocToggle_' + id] = function(trigger) {
                var dropdown = document.getElementById('tocDropdown_' + id);
                if (dropdown.classList.contains('open')) {
                    window['tocClose_' + id]();
                } else {
                    dropdown.classList.add('open');
                    trigger.classList.add('active');
                    trigger.setAttribute('aria-expanded', 'true');
                }
            };

            // Close dropdown
            window['tocClose_' + id] = function() {
                var dropdown = document.getElementById('tocDropdown_' + id);
                var trigger = document.querySelector('.toc-fab-' + id);
                if (dropdown) dropdown.classList.remove('open');
                if (trigger) {
                    trigger.classList.remove('active');
                    trigger.setAttribute('aria-expanded', 'false');
                }
            };

            // Scroll to top of page
            window['tocScrollToTop_' + id] = function() {
                window.scrollTo({ top: 0, behavior: 'smooth' });
                window['tocClose_' + id]();
            };

            // Scroll to section with offset for fixed header
            window['tocScrollTo_' + id] = function(targetId) {
                var target = document.getElementById(targetId);
                if (target) {
                    var targetPosition = target.getBoundingClientRect().top + window.pageYOffset - getScrollOffset();
                    window.scrollTo({ top: targetPosition, behavior: 'smooth' });
                }
                window['tocClose_' + id]();
            };

            // Close on outside click
            document.addEventListener('click', function(e) {
                if (!e.target.closest('#tocFabWrapper_' + id)) {
                    window['tocClose_' + id]();
                }
            });

            // Position on load and resize
            positionFab();
            window.addEventListener('resize', positionFab);
        })();
