    function search(criteriaType) {

        $(".loader").addClass("is-active");

        var isValid = true;
        var action = "";
        var addressLine2 = "";
        var criteriaString = "";
        var city = "";
        var state = "";

        switch (criteriaType) {
        case "SearchByName":
        case "SearchByNameAndState":
        case "SearchByNameCityAndState":

                var firstName = $("#SearchCriteriaViewModel_FirstName").val().trim();
                var middleName = $("#SearchCriteriaViewModel_MiddleName").val().trim();
                var lastName = $("#SearchCriteriaViewModel_LastName").val().trim();
                addressLine2 = $("#SearchByName_AddressLine2").val().trim();

                if (lastName.length == 0) {

                    $(".alert").fadeIn();
                    $(".alert").html(
                        "Please supply at least a last name. For the best results, include a first name, last name along with the city and state");

                    $(".loader").removeClass("is-active");

                    return false;
                }

           if (middleName.length > 0 && firstName.length == 0) {

               $(".alert").fadeIn();
               $(".alert").html(
                   "Please supply a first and last name when including a middle initial");

               $(".loader").removeClass("is-active");

               return false;
           }

            if (firstName.length > 0 && lastName.length == 0) {

                $(".alert").fadeIn();
                $(".alert").html(
                    "Please supply a last name when including a first name");

                $(".loader").removeClass("is-active");

                return false;
                }

            criteriaString = firstName + "-" + middleName + "-" + lastName;
            criteriaString = criteriaString.replace(/ /g, "-").replace(/--/, "-").replace(/'/, "").replace(/,/, "")
                .trim();

            if (addressLine2.length > 0) {

                //Do we have a city/state
                city = addressLine2.substring(0, addressLine2.indexOf(",")).trim();
                state = addressLine2.substring(addressLine2.indexOf(",") + 1).trim();

                if (state.length > 0) {
                    criteriaString = criteriaString + "/" + state;
                }

                if (city.length > 0) {
                    criteriaString = criteriaString + "/" + city;
                }

                criteriaString = criteriaString.replace(/ /g, "-").replace(/--/, "-").replace(/'/, "")
                    .replace(/,/, "")
                    .replace(/,/, "-").trim();

                if ((city.length > 0 || state.length > 0) && lastName.length == 0) {

                    $(".alert").fadeIn();
                    $(".alert").html(
                        "Please supply a last name when including a state or city/state");

                    $(".loader").removeClass("is-active");

                    return false;
                }
            }

            criteriaString = trimStart("-", criteriaString);
                action = "/people/" + criteriaString.toLowerCase();

            break;
        case "SearchByAddress":

            try {

                var addressLine1 = $("#SearchCriteriaViewModel_AddressLine1").val().trim();
                addressLine2 = $("#SearchByAddress_AddressLine2").val().trim();

                if (addressLine1.length == 0 || addressLine2.length == 0) {
                    isValid = false;
                }

                criteriaString = addressLine1.replace(/--/, "-").replace(/-/g, "~").replace(/ /g, "-").replace(/'/, "")
                    .replace(/,/g, "").replace(/\./g, "")
                    .trim();

                //Do we have a city/state
                city = addressLine2.substring(0, addressLine2.indexOf(",")).trim();
                state = addressLine2.substring(addressLine2.indexOf(",") + 1).trim();

                if (city.length > 0) {
                    criteriaString = criteriaString + "/" + city;
                } else {
                    isValid = false;
                }

                if (state.length > 0) {
                    criteriaString = criteriaString + "/" + state;
                } else {
                    isValid = false;
                }

                criteriaString = criteriaString.replace(/ /g, "-").replace(/--/, "-").replace(/'/, "")
                    .replace(/,/, "")
                    .replace(/,/, "-").trim();

                criteriaString = trimStart("-", criteriaString);
                action = "/address/" + criteriaString.toLowerCase();

                if (!isValid) {

                    $(".alert").fadeIn();
                    $(".alert").html(
                        "Please supply a valid address");

                    $(".loader").removeClass("is-active");

                    return false;
                }

            } catch (exception) {

                $(".alert").fadeIn();
                $(".alert").html(
                    "Unsupported inputs entered");

                $(".loader").removeClass("is-active");

                return false;
            }

            break;
        case "SearchByPhone":

                var phone = $("#SearchCriteriaViewModel_Phone").val().replace(/ /g, "-").replace("(", "").replace(")", "")
                    .trim();

                if (phone.length < 10) {
                    isValid = false;
                }

            if (!isValid) {

                $(".alert").fadeIn();
                $(".alert").html(
                    "Please supply a valid 10 digit phone number");

                $(".loader").removeClass("is-active");

                return false;
            }

            action = "/phone/" + phone;

            break;
        case "SearchByEmail":

                var email = $("#Email").val().trim();

            if (!/^\w+([\.-]?\w+)*@\w+([\.-]?\w+)*(\.\w{2,3})+$/.test(email))
            {
                isValid = false;
            }

            if (!isValid) {

                $(".alert").fadeIn();
                $(".alert").html(
                    "Please supply a valid email address");

                $(".loader").removeClass("is-active");

                return false;
            }

                action = "/email/" + email.replace("@", "_.");

            break;
            default:

                $(".loader").removeClass("is-active");

                return false;
        }

        window.location.href = action;

        return true;
    }

    function trimStart(character, string) {
        var startIndex = 0;

        while (string[startIndex] === character) {
            startIndex++;
        }

        return string.substr(startIndex);
    }
