(function() {
    var banner = document.getElementById('universalPwaInstallBanner');
    var bannerText = document.getElementById('universalPwaInstallBannerText');
    var instructionsPanel = document.getElementById('universalPwaInstallInstructionsPanel');
    var instructionsTitle = document.getElementById('universalPwaInstructionsTitle');
    var instructionsContent = document.getElementById('universalPwaInstructionsContent');
    var overlay = document.getElementById('universalPwaInstallOverlay');
    var installButton = document.getElementById('universalPwaInstallBtn');
    var dismissButton = banner.querySelector('.universal-pwa-dismiss-btn');
    var closeInstructionsButton = instructionsPanel.querySelector('.universal-pwa-close-instructions-btn');

    var cookieName = "universalPwaInstallDismissed";
    var dismissDurationDays = parseInt("7", 10);
    var initialDisplayBanner = false;
    var androidAppStoreUrl = "https://play.google.com/store/apps/details?id=com.freedataservices.truepeoplesearch&utm_source=tps&utm_campaign=home&utm_medium=web";
    var androidAppName = "TruePeopleSearch";
    var appName = "TruePeopleSearch";
    
    // Platform-specific messages
    var iosMessage = "Install our app, no app store required!";
    var androidMessage = "Get the TruePeopleSearch app!";
    var iosButtonText = "Install";
    var androidButtonText = "Install";

    // Check if user came from PWA and set cookie if needed
    checkAndSetPWAAppCookie();
    
    // Device detection
    var deviceInfo = getDeviceInfo();

    function checkAndSetPWAAppCookie() {
        // Check if URL contains utm_source=pwa
        var urlParams = new URLSearchParams(window.location.search);
        var utmSource = urlParams.get('utm_source');
        
        if (utmSource === 'pwa') {
            // Check if ispwaapp cookie doesn't exist
            if (!getCookie('ispwaapp')) {
                // Set the cookie to indicate user is in PWA app
                setCookie('ispwaapp', '1', 365); // Set for 1 year
            }
        }
    }

    function getDeviceInfo() {
        var userAgent = navigator.userAgent.toLowerCase();
        var isIOS = /iphone|ipad|ipod/.test(userAgent);
        var isAndroid = /android/.test(userAgent);
        var isSafari = isIOS && !/crios|fxios|edgios/.test(userAgent) && /version\/[\d\.]+.*safari/.test(userAgent);
        var isInStandaloneMode = window.matchMedia('(display-mode: standalone)').matches || navigator.standalone === true;
        var isAlreadyPWA = document.cookie.indexOf('ispwaapp=1') > -1 || window.location.href.indexOf('utm_source=pwa') > -1;

        return {
            isIOS: isIOS,
            isAndroid: isAndroid,
            isSafari: isSafari,
            isInStandaloneMode: isInStandaloneMode,
            isAlreadyPWA: isAlreadyPWA,
            deviceType: isIOS ? 'ios' : (isAndroid ? 'android' : 'other')
        };
    }

    function getCookie(name) {
        var nameEQ = name + "=";
        var ca = document.cookie.split(';');
        for(var i=0;i < ca.length;i++) {
            var c = ca[i];
            while (c.charAt(0)==' ') c = c.substring(1,c.length);
            if (c.indexOf(nameEQ) == 0) return c.substring(nameEQ.length,c.length);
        }
        return null;
    }

    function setCookie(name, value, days) {
        var expires = "";
        if (days) {
            var date = new Date();
            date.setTime(date.getTime() + (days*24*60*60*1000));
            expires = "; expires=" + date.toUTCString();
        }
        document.cookie = name + "=" + (value || "")  + expires + "; path=/";
    }

    function shouldShowPrompt() {
        if (getCookie(cookieName)) return false;
        
        if (deviceInfo.deviceType === 'ios') {
            return deviceInfo.isSafari && !deviceInfo.isInStandaloneMode && !deviceInfo.isAlreadyPWA;
        } else if (deviceInfo.deviceType === 'android') {
            return !deviceInfo.isInStandaloneMode && !deviceInfo.isAlreadyPWA;
        }
        
        return false;
    }

    function updateBannerForDevice() {
        if (deviceInfo.deviceType === 'ios') {
            bannerText.textContent = iosMessage;
            installButton.textContent = iosButtonText;
        } else if (deviceInfo.deviceType === 'android') {
            bannerText.textContent = androidMessage;
            installButton.textContent = androidButtonText;
        }
    }

    function createIOSInstructions() {
        var imgShare = "/images/ios-share-icon.png";
        var imgAddToHome = "/images/add-to-home.png";
        var imgAdd = "/images/add-to-home.png";
        
        return '<p>Install TruePeopleSearch on your home screen, no app store required:</p>' +
               '<div class="instruction-step">' +
               '<img src="' + imgShare + '" alt="Safari Share Icon" loading="lazy" />' +
               '<span><strong>1.</strong> Tap the <strong>Share</strong> button below.</span>' +
               '</div>' +
               '<div class="instruction-step">' +
               '<img src="' + imgAddToHome + '" alt="Add to Home Screen Option" loading="lazy" />' +
               '<span><strong>2.</strong> Scroll down and tap <strong>"Add to Home Screen"</strong>.</span>' +
               '</div>' +
               '<div class="instruction-step">' +
               '<img src="' + imgAdd + '" alt="Add Button" loading="lazy" />' +
               '<span><strong>3.</strong> Tap <strong>"Add"</strong> in the top right corner.</span>' +
               '</div>';
    }



    function showBanner() {
        if (banner) {
            updateBannerForDevice();
            banner.classList.remove('universal-pwa-widget-hidden');
        }
    }

    function hideBanner() {
        if (banner) {
            banner.classList.add('universal-pwa-widget-hidden');
        }
    }

    function showInstructions() {
        // Only show instructions for iOS - Android goes directly to Play Store
        if (!instructionsPanel || !overlay || deviceInfo.deviceType !== 'ios') return;
        
        hideBanner();
        
        // Set up iOS PWA installation content
        instructionsTitle.textContent = 'Install ' + appName;
        instructionsContent.innerHTML = createIOSInstructions();
        
        overlay.classList.add('visible');
        instructionsPanel.classList.add('visible');
    }

    function hideInstructions(showBannerAfter) {
        if (!instructionsPanel || !overlay) return;

        instructionsPanel.classList.remove('visible');
        overlay.classList.remove('visible');
        if (showBannerAfter && initialDisplayBanner && !getCookie(cookieName)) {
            showBanner();
        }
    }
    
    function dismissPrompt() {
        setCookie(cookieName, 'true', dismissDurationDays);
        hideBanner();
        if (instructionsPanel) instructionsPanel.classList.remove('visible');
        if (overlay) overlay.classList.remove('visible');
    }

    function handleInstallClick() {
        if (deviceInfo.deviceType === 'ios') {
            showInstructions();
        } else if (deviceInfo.deviceType === 'android') {
            window.open(androidAppStoreUrl, '_blank', 'noopener');
            dismissPrompt(); // Auto-dismiss after redirecting to app store
        }
    }

    if (shouldShowPrompt()) {
        // Expose function to show instructions globally
        window.showUniversalPWAInstallInstructions = function() {
            showInstructions();
        };

        if (initialDisplayBanner && banner) {
            showBanner();

            installButton.addEventListener('click', function(e) {
                e.stopPropagation(); 
                handleInstallClick();
            });
            
            banner.addEventListener('click', function(e) {
                if (e.target !== dismissButton && e.target.parentNode !== dismissButton && e.target !== installButton) {
                    handleInstallClick();
                }
            });

            dismissButton.addEventListener('click', function(e) {
                e.stopPropagation();
                dismissPrompt();
            });
        }

        // Panel event listeners
        if (closeInstructionsButton) {
            closeInstructionsButton.addEventListener('click', function() {
                hideInstructions(true); 
            });
        }
        
        if (overlay) {
            overlay.addEventListener('click', function() {
                hideInstructions(true);
            });
        }
    }
})();
