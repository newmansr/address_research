function updateLinkHrefs(container) {
  const links = container.querySelectorAll('a');

  links.forEach(link => {
    if (!link.href) return;

    try {
      const url = new URL(link.href);
      const u1 = url.searchParams.get('u1');

      if (u1 && u1.includes('mobile')) {
        url.searchParams.set('u1', u1.replace('mobile', 'desktop'));
        link.href = url.toString();
      }
    } catch (e) {
      // skip malformed URLs
    }
  });
}
