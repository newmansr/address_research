    function search(criteriaType) {

        var city = "";

                criteriaString = criteriaString.replace(/ /g, "-").replace(/--/, "-").replace(/'/, "")
                    .replace(/,/, "")
                    .replace(/,/, "-").trim();

                if ((city.length > 0 || state.length > 0) && lastName.length == 0) {

                    $(".alert").fadeIn();
                    $(".alert").html(
                        "Please supply a last name when including a state or city/state");

                    $(".loader").removeClass("is-active");

                    return false;
                }
            }

            criteriaString = trimStart("-", criteriaString);
                action = "/people/" + criteriaString.toLowerCase();

            break;

                    .replace(/,/, "")
                    .replace(/,/, "-").trim();

                    return false;

            } catch (exception) {

                $(".alert").fadeIn();
                $(".alert").html(
                    "Unsupported inputs entered");

                $(".loader").removeClass("is-active");

                return false;

    function trimStart(character, string) {

            if (typeof $ === 'undefined' || typeof $.fn.autoComplete === 'undefined') return;

            var autocompleteConfig = {
                minLength: 2,
                delay: 500,
                autoSelect: true,
                resolver: 'city_state_resolver',
                noResultText: 'No suggestions',
                events: {
                    search: function (text, callback) {
                        if (text != null && text.length > 1) {
                            $.ajax({
                                method: "POST",
                                url: "/api/Data/GetCityStateSuggestions",
                                data: { input: text }
                            })
                            .done(function (result) { callback(result); })
                            .fail(function () {});
                        }
                    }
                },
                formatResult: function (suggestion) { return suggestion.suggestText; }

                });
