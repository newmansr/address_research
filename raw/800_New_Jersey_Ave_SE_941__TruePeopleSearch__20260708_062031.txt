    function getDeviceId() {
        var key = "_tps_did";
        var id = null;
        try { id = localStorage.getItem(key); } catch (e) { }
        if (!id || !/^[0-9a-f]{40,128}$/.test(id)) {
            var arr = new Uint8Array(32);
            crypto.getRandomValues(arr);
            id = Array.prototype.map.call(arr, function (b) {
                return ("0" + b.toString(16)).slice(-2);
            }).join("");
            try { localStorage.setItem(key, id); } catch (e) { }
        }
        return id;
    }

    // ── API helper (calls our proxy which adds the API key) ─────────────────
    function surveyApi(action, body, callback) {
        var did = getDeviceId();
        var url = (action === "start")
            ? "/survey/start"
            : "/survey/" + action + "/" + did;

        var xhr = new XMLHttpRequest();
        xhr.open("POST", url, true);
        xhr.setRequestHeader("Content-Type", "application/json");
        xhr.onreadystatechange = function () {
            if (xhr.readyState === 4) {
                var data = null;
                try { data = JSON.parse(xhr.responseText); } catch (e) { data = {}; }
                if (callback) callback(data, xhr.status);
            }
        };
        xhr.send(JSON.stringify(body || {}));
    }
