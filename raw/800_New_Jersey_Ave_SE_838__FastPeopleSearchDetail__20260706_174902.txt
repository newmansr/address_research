  links.forEach(link => {

    try {
      const url = new URL(link.href);

        url.searchParams.set('u1', u1.replace('mobile', 'desktop'));
        link.href = url.toString();
      }
    } catch (e) {
      // skip malformed URLs
    }
  });
}
